# 大规模文本风格迁移系统 - 完整解决方案

## 一、系统架构设计

### 整体流程

```
源校史文档
    ↓
[阶段1] 预处理与风格提取
    ├─ 文档切分（按章节/段落）
    ├─ 风格标签提取（多维度）
    ├─ 风格示例片段保存
    └─ 建立索引（主题向量 + 风格向量）
    ↓
风格数据库（带ID）
    ↓
[阶段2] 目标校史改写
    ├─ 目标内容分段
    ├─ 语义检索匹配（找相似风格片段）
    ├─ 分批改写（带风格约束）
    ├─ 事实一致性检查
    └─ 质量评估与人工审核标记
    ↓
最终改写结果
```

```python
from docx import Document
import re

def split_document(file_path, min_length=200, max_length=400):
    """
    将文档切分为片段
    
    Returns:
        List[Dict]: 每个片段包含：
            - id: 唯一标识（如 "chapter1_para3"）
            - text: 文本内容
            - chapter: 所属章节
            - position: 在章节中的位置
            - main_title: 章节大标题
    """
    doc = Document(file_path)
    fragments = []
    current_chapter = "引言"
    current_chapter_id = 0
    para_in_chapter = 0
    
    for para in doc.paragraphs:
        text = para.text.strip()
        if not text:
            continue
        
        # 检测是否为标题（样式判断或格式判断）
        if para.style.name.startswith('Heading'):
            current_chapter = text
            current_chapter_id += 1
            para_in_chapter = 0
            continue
        
        # 切分为合适长度的片段
        if len(text) >= min_length:
            # 直接作为一个片段
            para_in_chapter += 1
            fragments.append({
                'id': f"chapter{current_chapter_id}_para{para_in_chapter}",
                'text': text,
                'chapter': current_chapter,
                'chapter_id': current_chapter_id,
                'position': para_in_chapter,
                'main_title': current_chapter
            })
        else:
            # 合并短段落
            # ...（合并逻辑）
    
    return fragments
```

### 3. 建立索引

**两层索引结构**：

```python
import faiss
import numpy as np
from typing import List, Dict

class StyleIndex:
    """风格索引系统"""
    
    def __init__(self):
        self.fragments = []  # 存储所有片段
        self.semantic_index = None  # FAISS 索引（语义相似度）
        self.topic_index = {}  # 主题倒排索引
        
    def add_fragment(self, fragment_data: Dict):
        """添加片段到索引"""
        self.fragments.append(fragment_data)
        
        # 添加到语义索引
        vector = np.array([fragment_data['semantic_vector']], dtype='float32')
        if self.semantic_index is None:
            self.semantic_index = faiss.IndexFlatL2(vector.shape[1])
        self.semantic_index.add(vector)
        
        # 添加到主题索引
        for tag in fragment_data['topic_tags']:
            if tag not in self.topic_index:
                self.topic_index[tag] = []
            self.topic_index[tag].append(fragment_data['fragment_id'])
    
    def search_similar(self, query_vector, top_k=3, threshold=0.75):
        """
        检索最相似的片段
        
        Args:
            query_vector: 查询向量
            top_k: 返回前K个结果
            threshold: 相似度阈值
            
        Returns:
            List[Dict]: 相似片段列表，包含相似度分数
        """
        query = np.array([query_vector], dtype='float32')
        distances, indices = self.semantic_index.search(query, top_k)
        
        results = []
        for dist, idx in zip(distances[0], indices[0]):
            similarity = 1 / (1 + dist)  # 转换为相似度
            if similarity >= threshold:
                fragment = self.fragments[idx].copy()
                fragment['similarity'] = similarity
                results.append(fragment)
        
        return results
```

### 2. 智能检索匹配

```python
def find_matching_styles(target_fragment, style_index, top_k=3):
    """
    为目标片段找到最匹配的风格参考
    
    Args:
        target_fragment: 目标片段
        style_index: 风格索引
        top_k: 返回前K个匹配
        
    Returns:
        Dict: 匹配结果，包含风格标签和示例片段
    """
    # 1. 语义相似度检索
    similar_fragments = style_index.search_similar(
        target_fragment['semantic_vector'],
        top_k=top_k,
        threshold=0.75
    )
    
    # 2. 主题标签匹配（补充）
    topic_matches = []
    for tag in target_fragment['topic_tags']:
        if tag in style_index.topic_index:
            topic_matches.extend(style_index.topic_index[tag])
    
    # 3. 合并结果
    if similar_fragments:
        # 有高相似度匹配
        best_match = similar_fragments[0]
        return {
            'match_type': 'high_similarity',
            'confidence': best_match['similarity'],
            'style_labels': best_match['style_labels'],
            'quantitative_features': best_match['quantitative_features'],
            'example_fragments': [f['text'] for f in similar_fragments[:2]],
            'reference_ids': [f['fragment_id'] for f in similar_fragments]
        }
    elif topic_matches:
        # 有主题匹配
        matched_fragment = style_index.fragments[topic_matches[0]]
        return {
            'match_type': 'topic_match',
            'confidence': 0.6,
            'style_labels': matched_fragment['style_labels'],
            'example_fragments': [matched_fragment['text']],
            'reference_ids': [matched_fragment['fragment_id']]
        }
    else:
        # 使用全局风格（兜底）
        return {
            'match_type': 'global_fallback',
            'confidence': 0.5,
            'style_labels': style_index.global_style['avg_style_scores'],
            'example_fragments': style_index.global_style['example_paragraphs'],
            'reference_ids': []
        }
```

### 3. 分批改写（带风格约束）

```python
def batch_rewrite(target_fragments, style_index, batch_size=5):
    """
    分批改写目标片段
    
    Args:
        target_fragments: 目标片段列表
        style_index: 风格索引
        batch_size: 每批处理的片段数
        
    Returns:
        List[Dict]: 改写结果列表
    """
    results = []
    
    for i in range(0, len(target_fragments), batch_size):
        batch = target_fragments[i:i+batch_size]
        
        for fragment in batch:
            # 1. 找到匹配的风格
            style_match = find_matching_styles(fragment, style_index)
            
            # 2. 调用改写器
            rewritten = rewrite_with_style(
                target_text=fragment['text'],
                style_labels=style_match['style_labels'],
                example_fragments=style_match['example_fragments'],
                quantitative_features=style_match['quantitative_features']
            )
            
            # 3. 事实一致性检查
            fact_check = check_fact_consistency(
                original=fragment['text'],
                rewritten=rewritten
            )
            
            # 4. 保存结果
            result = {
                'fragment_id': fragment['id'],
                'original_text': fragment['text'],
                'rewritten_text': rewritten,
                'style_match': style_match,
                'fact_check': fact_check,
                'needs_review': not fact_check['fact_consistent']
            }
            
            results.append(result)
    
    return results
```

```python
def check_fact_consistency(original, rewritten, llm_client):
    """
    检查改写后是否保持事实一致
    
    Args:
        original: 原始文本
        rewritten: 改写后文本
        llm_client: LLM 客户端
        
    Returns:
        Dict: 包含一致性判断和差异列表
    """
    prompt = f"""请对比以下两段文本，判断改写后是否保留了所有关键事实（时间、人物、事件、数据），是否添加了原文没有的虚构内容。

原文（目标校史原始素材）：
{original}

改写后：
{rewritten}

请输出 JSON 格式：
{{
  "fact_consistent": true/false,
  "key_facts_preserved": ["事实1", "事实2"],
  "missing_facts": ["缺失的事实"],
  "added_content": ["虚构的内容"],
  "differences": ["差异描述"]
}}
"""
    
    response = llm_client.invoke(prompt)
    result = parse_json_response(response)
    
    return result
```

### 5. 质量评估与人工审核标记

```python
def quality_assessment(rewrite_results):
    """
    评估改写质量并标记需要人工审核的片段
    
    Args:
        rewrite_results: 改写结果列表
        
    Returns:
        Dict: 包含质量报告和审核队列
    """
    review_queue = []
    quality_report = {
        'total_fragments': len(rewrite_results),
        'high_quality': 0,
        'needs_review': 0,
        'low_quality': 0
    }
    
    for result in rewrite_results:
        # 判断质量等级
        if result['fact_check']['fact_consistent'] and result['style_match']['confidence'] >= 0.8:
            result['quality_level'] = 'high'
            quality_report['high_quality'] += 1
        elif not result['fact_check']['fact_consistent'] or result['style_match']['confidence'] < 0.6:
            result['quality_level'] = 'low'
            quality_report['low_quality'] += 1
            review_queue.append(result)
        else:
            result['quality_level'] = 'medium'
            quality_report['needs_review'] += 1
    
    return {
        'quality_report': quality_report,
        'review_queue': review_queue,
        'all_results': rewrite_results
    }
```
